class Solution {
    public int countDigits(int num) {
        int count = 0;
        int temp = num;

        while (temp > 0) {
            int digit = temp % 10;   // Extract last digit
            if (digit != 0 && num % digit == 0) {  
                // Check if digit divides num (avoid division by zero)
                count++;
            }
            temp /= 10;  // Remove last digit
        }

        return count;
    }
}
